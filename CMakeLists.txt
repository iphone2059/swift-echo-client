cmake_minimum_required(VERSION 3.24)
project(swift_echo_client_verification LANGUAGES NONE)
include(CTest)
if(NOT SWIFT_CONFIGURATION MATCHES "^(debug|release)$" OR NOT IS_DIRECTORY "${SWIFT_BIN_PATH}")
  message(FATAL_ERROR "Pass the built Swift configuration and product directory")
endif()
find_program(PWSH_EXECUTABLE NAMES pwsh REQUIRED)
function(swift_gate name script)
  add_test(NAME "${name}" COMMAND "${PWSH_EXECUTABLE}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/Tests/${script}" ${ARGN})
  set_tests_properties("${name}" PROPERTIES TIMEOUT 300 RUN_SERIAL TRUE WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endfunction()
swift_gate(swift-unit cec_unit_tests.ps1 -Configuration "${SWIFT_CONFIGURATION}")
swift_gate(source-policy cec_source_policy.ps1 )
swift_gate(ownership-features cec_ownership_compile_tests.ps1 -Configuration "${SWIFT_CONFIGURATION}" -Phase features)
swift_gate(ownership-owners cec_ownership_compile_tests.ps1 -Configuration "${SWIFT_CONFIGURATION}" -Phase owners)
swift_gate(memory-safety cec_memory_safety_checks.ps1 -BinPath "${SWIFT_BIN_PATH}" -Configuration "${SWIFT_CONFIGURATION}")
swift_gate(gate-regressions cec_verification_gate_tests.ps1 -Configuration "${SWIFT_CONFIGURATION}")
swift_gate(fault-processes cec_fault_process_tests.ps1 -DriverPath "${SWIFT_BIN_PATH}/swift-echo-client-fault-driver.exe")
swift_gate(process-contract cec_process_tests.ps1 -ClientPath "${SWIFT_BIN_PATH}/swift-echo-client.exe")
swift_gate(extended-process-contract cec_extended_process_tests.ps1 -ClientPath "${SWIFT_BIN_PATH}/swift-echo-client.exe" -DriverPath "${SWIFT_BIN_PATH}/swift-echo-client-fault-driver.exe")
swift_gate(reset-storm cec_reset_storm_tests.ps1 -ClientPath "${SWIFT_BIN_PATH}/swift-echo-client.exe")
if(SWIFT_CONFIGURATION STREQUAL "release")
  swift_gate(optimized-hot-path cec_optimization_gate.ps1 -BinPath "${SWIFT_BIN_PATH}")
endif()
